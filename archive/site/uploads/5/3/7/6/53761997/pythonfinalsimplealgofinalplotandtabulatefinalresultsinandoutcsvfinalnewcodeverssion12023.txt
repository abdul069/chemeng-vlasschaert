#---------------------------------------------------------------------------
# fig :
#---------------------------------------------------------------------------
#            uA      uB
# ->         ->      ->
#  -------° ------°-------°
#         1   A   2   B   3
# u=0     p1      p2      p3

# ref :https://www.youtube.com/watch?v=xPnzJ4gt9eE

# fig : {uA,uB},pi, i=1,2,3

# my website
# new:10/02/2023:maxima   : fvm : example 3:1d momentum equation,
#                                 1d continuity equation,pipeflow,version 1.0
# youtube : https://www.youtube.com/watch?v=N85HmIIgm4s&t=1333s
# download  (maxima,code)       :maximafvmexample3moment1dcont1dversion12023
#---------------------------------------------------------------------------
rho = 1 # density
AA = 3  # cross-section
AB = 1  # cross-section
p1 = 28 # pressure given, no correction
p3 = 0  # pressure given, no correction
alpha = 0.5  # relaxation factor (under relaxation)
# why pressure difference is higher.
nu = 100 # number of iterations
lp = 200 # number of loops
# initial guesses for uA and uB and p2
uA = np.array([5/3])
uB = np.array([5])
p2 = 12

# initialize matrix to store results
results = np.zeros((nu, 4))
counter = 0 # reduce matrix :results 
# start overall loop, until convergence
# F = rho*u*cross-section
for loop in range(lp):
    uA_old = uA
    uB_old = uB
    for i in range(1, nu):
        # maxima code
        # u[A]·F2=-A(A)·(p2-p1)
        uA_new = (p1 - p2) * AA / (rho * uA_old * AA)
        # u[B]·F3-u[A]·F2=-A(B)·(p3-p2)
        uB_new = ((rho * uA_old * AA) * uA_old + (p2 - p3) * AB) / (rho * uB_old * AB)
        uA_old = uA_old + (uA_new - uA_old) * alpha 
        uB_old = uB_old + (uB_new - uB_old) * alpha
        if (abs(uB_old - uB) + abs(uA_old - uA)).all() < 0.001:
            break
    uA = uA_old
    uB = uB_old
    # check convergence F(A)=F(B),F = rho*u*cross-section
    if np.all(abs(rho * uA * AA - rho * uB * AB) < 0.01):
        break
    # use velocities to get pressure corrections
    dA = (rho * uA)**(-1)
    dB = (rho * uB)**(-1)
    # pressure correction equation
    # maxima
    # p2^∄=(A(B)·u[B]^°-A(A)·u[A]^°)/(A(B)·d[B]+A(A)·d[A])
    p2p = (uA * AA - uB * AB) / (dA * AA + dB * AB)
    # p2 = p2 + p2p * alpha
    p2 += p2p * alpha
    # correction for velocities, given values p = 0 here
    # maxima
    # uA = uA + dA * (p1 - p2)
    uA = uA + dA * (0 - p2p)
    # maxima
    # uB = uB + dB * (p2 - p3)
    uB = uB + dB * (p2p - 0)
# ----------------------------------------------------------    
# how find what to use : dA & dB =dx
# ----------------------------------------------------------
# =>  units :
#-----------------------------------------------------------
# [m/s] = dx*[N/m^2] = dx*[(kg*m/s^2)/m^2] =dx*[kg/(m*s^2)]
# 1/dx =[kg/(m^2*s)] =[u*rho] = [m/s*kg/m^3] = [kg/(m^2*s)]
#-----------------------------------------------------------
# x=A,B
#----------------------------------------------------------
    # store results in matrix
    counter += 1
    results[loop, 0] = loop
    results[loop, 1] = uA[0]
    results[loop, 2] = uB[0]
    results[loop, 3] = p2
print(results)    
# remove values after cvg     
results = results[:counter, :]
