class vec2d():
# vector 2 dimensional plane (xy-plane)    
    def __init__(self,x,y): # don't forget this
        self.x = x
        self.y = y
# add vector 1 + vector 2 ( use def to this,see : quadratic
# def function ,need return function.
    def add(self,ve):
        return vec2d(self.x+ve.x,self.y+ve.y)
    def minus(self,ve):
        return vec2d(self.x-ve.x,self.y-ve.y)
# ve internal add vector = vector 2    

print( "vector 2d = vector two dimensional ")
print( "-----------------------------------")
print( "property: add two vectors ,vec2d   ")

print( " first vector = vec2d(2,3) " )

# v1 = object and x,y = method = attribute
# example : find y coordinate vec2d(x,y)
print("y coordinate vec2d(x,y) = ",v1.y)
print( "-----------------------------------")
