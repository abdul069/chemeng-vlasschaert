def select_mycolor():
    color = askcolor()   
    myl =Label(win,text = "preferred color selected",bg =color[1],font="Times 12 bold",width = 36).grid(row=0,column=0)               
    myl1 =Label(win,text = color,font = "Timed 12 bold",width = 36,bg="yellow").grid(row=1,column=0)
    c1=color[1]
    myl3 =Label(win,text = "Hexadecimal Color Code = "+color[1],font = "Timed 12 bold",
	width = 36,bg="yellow").grid(row=4,column=0)    
   
def submit():  
   colorget = color_var.get()  # get value from Entry box
   if colorget != "":       
      myl2 =Label(win,text = "color",font = "Timed 12 bold",bg=colorget,width = 33).grid(row=3,column=1)      
   else:
      sub_btn['state']='disable'
      
def submit_reenter():    
    sub_btn['state']='active'  # set state of button
    color_var.set("")
    # clear Entry box
    
def random_color():
    # randomize the hex code,lamba used : command = lambda: 'Button'
    r = lambda: random.randint(0,255)
    color1_entry.set('#%02X%02X%02X' % (r(),r(),r()))    
    
 
btn=Button(win,text ='select color',command = select_mycolor,width = 36,font="Times 12 bold",
           bg="red").grid(row=2,column=0)

               font="Times 12 bold" )

sub_btn1=Button(win,text = 'Re_enter The Color', command = submit_reenter,width =36,
