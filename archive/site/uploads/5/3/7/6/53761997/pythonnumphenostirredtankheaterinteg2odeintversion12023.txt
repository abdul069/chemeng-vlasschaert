# Constants
V = 1.0  # Volume (m^3)
Cp = 4186.0  # Heat capacity (J/kg/K)
T0 = 25.0  # Initial temperature (°C)
F_in = 0.1  # Inlet flow rate (m^3/s)
rho = 1000.0  # Density (kg/m^3)

# Define the differential equations, 2 odes
# Energy Balance
def dTdt(T, t, Q, F):
    return (Q(t)/(V*Cp)) + (F(t)*(T0 - T))/V
# Material Balance
def dVdt(V, t, F):
    return F_in - F(t)

# Define the input profiles
def Ti(t):
    if t < 300.0:
        return T0 + 0.05*t
    else:
        return 40.0

def Q(t):
    if t < 150.0:
        return 5000.0
    elif t < 450.0:
        return 5000.0 - 10.0*(t-150.0)
    else:
        return 4550.0

def F(t):
    if t < 250.0:
        return 0.1 + 0.0002*t
    elif t < 500.0:
        return 0.15 - 0.0001*(t-250.0)
    else:
        return 0.12

# Initial conditions
T0 = 20.0  # Initial temperature (°C)
V0 = V  # Initial volume (m^3)
t_sim = 600
# Solve the differential equations
t_span = np.linspace(0.0, t_sim, t_sim+1)  # Time span to solve over (s)
sol = odeint(dTdt, T0, t_span, args=(Q, F))
T = sol[:, 0]

sol = odeint(dVdt, V0, t_span, args=(F,))
V = sol[:, 0]
# Plot the temperature, input profiles, and flow rates
fig, ax1 = plt.subplots(figsize=(8, 6))
color = 'tab:red'
ax1.set_xlabel('Time (s)')
ax1.set_ylabel('Temperature (°C)', color=color)
ax1.plot(t_span, T, color=color, label='Temperature')
ax1.tick_params(axis='y', labelcolor=color)

ax2 = ax1.twinx()
color = 'tab:green'
ax2.set_ylabel('Flow Rates (m^3/s)', color=color)
ax2.plot(t_span, [F(t) for t in t_span], color=color, label='Flow Rate')
ax2.tick_params(axis='y', labelcolor=color)

ax3 = ax1.twinx()
color = 'tab:blue'
ax3.spines['right'].set_position(('axes', 1.1))
ax3.set_ylabel('Heat Input (W)', color=color)
ax3.plot(t_span, [Q(t)*(T0 - Ti(t)) for t in t_span], color=color, label='Heat Input')
ax3.tick_params(axis='y', labelcolor=color)

# Generate table
time_outp =t_sim/60 
table1_data = []
table2_data = []
for i in range(0, len(t_span), int(time_outp)):# don't forget intt ,range
    time = t_span[i]
    t_val = T[i]
    table1_data.append([time, t_val])
    
    ti_val = Ti(time)
    q_val = Q(time)
