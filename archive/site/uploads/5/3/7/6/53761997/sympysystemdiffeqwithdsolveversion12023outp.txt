a = ? positive => False
sym.symbols('t, C1, C2') = <class 'tuple'>
Function =  x(t) y(t)
de1 = Eq(x(t) + 3*Derivative(x(t), t), 1)
de2 = Eq(-a*y(t) - 15*x(t) + Derivative(y(t), t) + 4, 1)
[Eq(x(t), -C1*(3*a + 1)*exp(-t/3)/45 + 1), Eq(y(t), C1*exp(-t/3) + C2*exp(a*t) - 12/a)]
