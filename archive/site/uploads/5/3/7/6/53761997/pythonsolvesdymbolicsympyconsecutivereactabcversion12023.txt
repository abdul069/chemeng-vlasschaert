# Define the variables and functions
t = symbols('t')
A = Function('A')(t)
B = Function('B')(t)
C = Function('C')(t)
k1, k2 = symbols('k1 k2')
a = symbols('a', positive=True)  # Assume a is positive

# Define the differential equations
eq1 = Eq(Derivative(A, t), -k1*A)
eq2 = Eq(Derivative(B, t), k1*A - k2*B)
eq3 = Eq(Derivative(C, t), k2*B)

# Solve the differential equations with the initial conditions
soln = dsolve((eq1, eq2, eq3), ics={A.subs(t, 0): a, B.subs(t, 0): 0, C.subs(t, 0): 0}, check=True)
print("soln = ",soln)
# Find the values of the constants of integration C1, C2, and C3
# C1, C2, C3 = symbols('C1 C2 C3')

A_soln = soln[0]
B_soln = soln[1]
C_soln = soln[2]

print(A_soln.rhs) 
print(B_soln.rhs)
print(C_soln.rhs)
print("=> A(t),B(t),C(t) ")
print("A(t)=",A_soln.rhs)
print("B(t)=",B_soln.rhs)
print("C(t)=",C_soln.rhs)
print(".......................................................................")
print("checkodesol((eq1,eq2,eq3), soln) =",checkodesol((eq1,eq2,eq3), soln)    )
print(".......................................................................")
# plot de A,B,C
t_range = (t, 0, 10)
A_soln_subs = A_soln.rhs.subs({a: 1, k1: 1, k2: 0.5})
B_soln_subs = B_soln.rhs.subs({a: 1, k1: 1, k2: 0.5})
C_soln_subs = C_soln.rhs.subs({a: 1, k1: 1, k2: 0.5})

p = plot_parametric((t, A_soln_subs), (t, B_soln_subs), (t, C_soln_subs), t_range,
