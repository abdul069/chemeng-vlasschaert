from math import sqrt  # cmath ,complex variable
class vec2d():
# vector 2 dimensional plane (xy-plane)    
    def __init__(self,x,y): # don't forget this
        self.x = x
        self.y = y        
# change ve ,part: 5 -> other
    def add(self,other):
        return vec2d(self.x+other.x,self.y+other.y)
# part 5 . minus -> sub , see : appendix ( simple math class )   
    def sub(self,other):
        return vec2d(self.x-other.x,self.y-other.y)
# doesn't change  
    def scale(self,scalar):
        return vec2d(scalar*self.x,scalar*self.y)        
# doesn't change    
    def __eq__(self,other):   # see: logic and,eq = equality
       return self.x == other.x and self.y == other.y
# part operator overloading
    def __add__(self,other):   
        return self.add(other)
    def __sub__(self,other):   
        return self.sub(other)
# other = SCALAR    
    def __mul__(self,scalar):   # multiplication =*,mul
        return self.scale(scalar)
    def __rmul__(self,scalar):  # scalar multiplication=*,rmul  
        return self.scale(scalar)
# representation vec2d(x,y)
    def __repr__(self):   
        return str([self.x,self.y])
# outside class.vec2d -> "working"
# inside  class.vec2d -> error : NameError
def length(self):         
    return sqrt(self.x*self.x+self.y*self.y)
def dot(self,other):
    return self.x*other.x + self.y*other.y
