#-------------------------------------------------------------------------------
# 1e) reaction : 2A = C ,forward = k[1],backwards = k[2]
# reaction : r1 ⇒ (2)*A→ C  ,k[1] ⇒ forward
# reaction : r2 ⇒ C   → 2*A ,k[2] ⇒ forward
# 2e) reaction : C = A+B ,forward = k[2],backwards = k[3]
# reaction : r3 ⇒ C→ A+B  , k[3] ⇒ forward
# reaction : r4 ⇒ A+B → C , k[4] ⇒ forward
#-------------------------------------------------------------------------------
# show in maxima
#-------------------------------------------------------------------------------
#rem : python index start by zero : k[i] in code k[i-1],see reaction,below
#-------------------------------------------------------------------------------
# Initial conditions
A0 = 2
B0 = 0.5
C0 = 0
t = np.linspace(0, 10, 1000)
# Rate constants
k = [1.2, 0.8, 0.3,1.4]
# Set the initial conditions for the differential equations
y0 = [A0, B0, C0]
#-----------------------------------------------------------
# reaction differential equations : use with of odeint
# error :reaction(t,y) => correct:reaction(y,t)
#-----------------------------------------------------------
def reaction(y,t):
    A, B, C = y
    dAdt = k[2] * C + 2 * k[1] * C - k[3] * A * B - 2 * k[0] * A**2
    dBdt = k[3] * A * B - k[2] * C
    dCdt = k[2] * C - k[1] * C + k[3] * A * B + k[0] * A**2
    return [dAdt, dBdt, dCdt]
solution = odeint(reaction, y0, t)
A, B, C = solution.T
print("solution ")
print(".........")
print(solution) # without time
