from scipy.integrate import odeint

#-------------------------------------------------------------------------------
# reaction kinetics:2A=C,C=A+B,version 1.0
# peter.vlasschaert@gmail.com,02/01/2023
# maxima : code >maximareactionkinetics2AequalCandCequalAplusBversion12023.wmx
#-------------------------------------------------------------------------------
# 1e) reaction : 2A = C ,forward = k[1],backwards = k[2]
# reaction : r1 ⇒ (2)*A→ C  ,k[1] ⇒ forward
# reaction : r2 ⇒ C   → 2*A ,k[2] ⇒ forward
# 2e) reaction : C = A+B ,forward = k[2],backwards = k[3]
# reaction : r3 ⇒ C→ A+B  , k[3] ⇒ forward
# reaction : r4 ⇒ A+B → C , k[4] ⇒ forward
#-------------------------------------------------------------------------------
# show in maxima
#-------------------------------------------------------------------------------
k1 = 0.1
k2 = 0.2
k3 = 0.3
k4 = 0.4
   
def reaction(y, t):
    A, B, C = y
    dAdt = k3 * C + 2 * k2 * C - k4 * A * B - 2 * k1 * A**2
    dBdt = k4 * A * B - k3 * C
    dCdt = k3 * C - k2 * C + k4 * A * B + k1 * A**2
    return [dAdt, dBdt, dCdt]
# time domain
ts=np.linspace(0, 100, 30)
# Initial conditions
A0 = 1
B0 = 1.2
C0 = 1

# vector initial condition
v0 = [A0, B0, C0]
# integrate
solution = odeint(reaction, v0, ts)
print(solution)
# plot
plt.plot(ts, solution[:, 0], label='A(t)');
plt.plot(ts, solution[:, 1], label='B(t)');
plt.plot(ts, solution[:, 2], label='C(t)');
plt.legend();
plt.xlabel('time');
plt.show()
