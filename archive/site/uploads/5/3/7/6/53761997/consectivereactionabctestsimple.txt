import matplotlib.pyplot as plt
from scipy.integrate import odeint
import numpy as np
# A > B > C, k[0],k[1]
# python ,consecutive_reaction (simple),version 1.0
# peter.vlasschaert@gmail.com,03/01/2023
def consecutive_reaction(init_conditions, t):
    A = init_conditions[0]
    B = init_conditions[1]
    C = init_conditions[2]

    k = [0.1, 0.2]
    
    dAdt = -k[0] * A  # Derivative of A with respect to time
    dBdt = k[0] * A - k[1] * B  # Derivative of B with respect to time
    dCdt = k[1] * B  # Derivative of C with respect to time
    return [dAdt, dBdt, dCdt]

# Initial conditions
A0 = 1.0
B0 = 0.0
C0 = 0.0

# Time points to evaluate the solution
t = np.linspace(0, 5, 100)

# Solve the differential equations
solution = odeint(consecutive_reaction, [A0, B0, C0], t)

# Extract the concentrations of each species
A = solution[:, 0]
B = solution[:, 1]
C = solution[:, 2]

# Plot the concentrations of each species
plt.plot(t, A, label='A')
plt.plot(t, B, label='B')
plt.plot(t, C, label='C')

# Add a legend and labels to the plot
plt.legend()
plt.xlabel('Time')
plt.ylabel('Concentration')
print( " close plot :to see final concentration A,B,C ")
# Show the plot
plt.show()
# Print the final concentrations of each species
print("to see :final concentration A,B,C ")
print("----------------------------------")
print(f'A = {A[-1]:.3f}')
print(f'B = {B[-1]:.3f}')
print(f'C = {C[-1]:.3f}')
