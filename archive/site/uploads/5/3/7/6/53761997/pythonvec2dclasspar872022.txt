class vec2d():
    counter = 0

    # vector 2 dimensional plane (xy-plane)
    def __init__(self, x, y):  # don't forget this
        # do first
        vec2d.counter = vec2d.counter + 1
        # do second
        self.x = x
        self.y = y

    # representation vec2d(x,y)
    def __repr__(self):
        return str([self.x, self.y])

    # create addNumbers static method
    @staticmethod
    def count():
        return vec2d.counter
