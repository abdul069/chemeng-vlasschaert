class vec2d():
# vector 2 dimensional plane (xy-plane)    
    def __init__(self,x,y): # don't forget this
        self.x = x
        self.y = y
# add vector 1 + vector 2 ( use def to this,see : quadratic
# def function ,need return function.
    def add(self,ve):
        return vec2d(self.x+ve.x,self.y+ve.y)
    def minus(self,ve):
        return vec2d(self.x-ve.x,self.y-ve.y)
#..................................
# ve internal add vector = vector 2
#..................................
    def scale(self,scalar):
        return vec2d(scalar*self.x,scalar*self.y)
#..................................
# scala*vector = result new vector
#..................................    
    def __eq__(self,other):   # see: logic and,eq = equality
       return self.x == other.x and self.y == other.y
#.......................................
# equality method,otherwise wrong result
#.......................................       
    def __repr__(self):      # see: repr = representation 
      return "vec2d({},{})".format(self.x,self.y)
#.......................................
# representation of vector : combination
#.......................................
